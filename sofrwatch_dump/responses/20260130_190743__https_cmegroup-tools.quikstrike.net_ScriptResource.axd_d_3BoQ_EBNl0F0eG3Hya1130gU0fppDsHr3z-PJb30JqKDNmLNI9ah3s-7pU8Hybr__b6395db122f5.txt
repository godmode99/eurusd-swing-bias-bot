Type.registerNamespace('QuikStrike.Highcharts');

$type = QuikStrike.Highcharts.ChartControl = function (element) {
    QuikStrike.Highcharts.ChartControl.initializeBase(this, [element]);

    this.jsonSettings = null;
    this.Settings = null;
    this.fnCreateChart = null;
    this.highchart = null;
    this.options = null;
}

$type.prototype =
{

    get_JSONSettings: function () {
        return this.jsonSettings;
    },

    set_JSONSettings: function (value) {
        if (this.jsonSettings != value) {
            this.jsonSettings = value;
            this.raisePropertyChanged('JSONSettings');
        }
    },

    initialize: function () {
        QuikStrike.Highcharts.ChartControl.callBaseMethod(this, 'initialize');

        this.Settings = $.parseJSON(this.jsonSettings);
        this.Settings.Id = this.get_id() /* NOTE get_id() is a method on the base class Sys.UI.Control */;

        Highcharts.setOptions({
            lang: this.Settings.Language
        });
    },

    initBase: function (options) {

        this.options = options;
        options.custom = {};
        options.custom.Settings = this.Settings;

        // initialize objects
        if (options.chart == undefined) options.chart = {};
        if (options.chart.events == undefined) options.chart.events = {};
        if (options.plotOptions == undefined) options.plotOptions = {};
        if (options.plotOptions.series == undefined) options.plotOptions.series = {};

        options.chart.events.render = this.chartRender;
        options.chart.events.load = this.chartLoad;
        options.chart.animation = !this.Settings.IsPdf
        options.plotOptions.series.animation = !this.Settings.IsPdf

        // add title and subtitle
        if (this.Settings.Title != null) {
            if (options.title == undefined) options.title = {};
            options.title.text = this.Settings.Title;
        }
        if (this.Settings.Subtitle != null) {
            if (options.subtitle == undefined) options.subtitle = {};
            options.subtitle.text = this.Settings.Subtitle;
        }

        this.addAxes(options);
        this.addSeries(options);
        this.setTooltipPositioner(options);
        this.setLegendFormatter(options);
        this.addLabelFormatters(options);
        this.setRangeSelector(options, this.Settings.RangeSelector);

        if (options.rangeSelector != null && this.Settings.DefaultRangeIndex != null)
            options.rangeSelector.selected = this.Settings.DefaultRangeIndex;

        this.addIndicators(options);
        this.addPlotLines(options);
        this.addPlotBands(options);
        this.addExporting(options);

        this.setLegendPosition(options);

        //if (this.Settings.Width != null)
        //    options.chart.width = this.Settings.Width;

    },

    createChart: function () {

        // Let me explain...
        // In some cases we have script that executes after the chart is created (i.e. "qs-vtabs") and when this happens the intial 
        // rendering of the chart might be the wrong size so we use our own ScriptLoaded event to ensure the chart is created after all of
        // our script has executed...however, for god knows why, when the view is rendered by the internal WebBrowser control for creating
        // twitter or askqs images the script loading occurs in a different order so we need to create the chart directly


        //$('#divDebugOutput').text('hello' + window.ABCpdf_go);
        //window.ABCpdf_go = false;
        //window.MyTest = 1;

        if (this.Settings.IsPdf)
            this.fnCreateChart();
        else
        {
            $(window).unbind('ScriptLoaded', this.fnCreateChart);
            $(window).bind('ScriptLoaded', this.fnCreateChart);
        }
    },

    setRangeSelector: function (options, rs) {

        if (rs == null)
            return;
        if (options.rangeSelector == undefined) options.rangeSelector = {};

        options.rangeSelector.enabled = rs.Enabled;
        options.rangeSelector.inputEnabled = rs.InputEnabled;

        options.rangeSelector.buttons = [];
        for (i = 0; i < rs.Ranges.length; i++) {
            options.rangeSelector.buttons.push({
                type: rs.Ranges[i].type,
                count: rs.Ranges[i].count,
                text: rs.Ranges[i].text
            });
        }
        options.rangeSelector.selected = rs.DefaultIndex;
        if (rs.TooltipFormat != null) {
            if (options.tooltip == undefined) options.tooltip = {};
            options.tooltip.xDateFormat = rs.TooltipFormat;
        }
    },

    updateLabel: function (labelId, msg) {
        if (labelId != null && labelId.length > 0 && $('#' + labelId).length > 0)
            $('#' + labelId).text(msg);
    },

    chartLoad: function (event) {
        var settings = this.userOptions.custom.Settings;

        $find(settings.Id).highchart = this;

        var redraw = false;
        var axes = settings.Axes;
        if (axes != null && axes.length > 0) {
            for (var i = 0; i < axes.length; i++) {
                if (axes[i].DefaultMin != null || axes[i].DefaultMax != null) {
                    redraw = true;
                    if (axes[i].Type == 'X')
                        this.xAxis[axes[i].AxisIndex].setExtremes(axes[i].DefaultMin, axes[i].DefaultMax, false);
                    else
                        this.yAxis[axes[i].AxisIndex].setExtremes(axes[i].DefaultMin, axes[i].DefaultMax, false);
                    this.showResetZoom();
                }
            }
        }

        if (redraw)
            this.redraw();

        if (settings.ChartLoad != null)
            settings.ChartLoad(this, event);
    },

    chartRender: function (event) {
        var func = this.userOptions.custom.Settings.ChartRender;
        if (func != null)
            func(this, event);
        if (this.userOptions.custom.Settings.WatermarkImg != null)
            hcbc_addWatermark(this);
    },

    addAxes: function (options) {
        var axes = this.Settings.Axes;
        if (axes == null || axes.length == 0)
            return;

        for (var i = 0; i < axes.length; i++) {
            var axis = hcbc_getAxis(axes[i].Type == 'X', axes[i].AxisIndex, options);
            $.extend(axis, axes[i]);
        }
    },

    addSeries: function(options) {
        var series = this.Settings.Series;
        if (series == null || series.length == 0)
            return;

        if (options.series == undefined) options.series = [];

        for (var i = 0; i < series.length; i++) {
            var s = series[i];
            options.series.push(s);
        }
    },

    addLabelFormatters: function(options) {
        var formatters = this.Settings.LabelFormatters;
        if (formatters == null || formatters.length == 0)
            return;

        for (var i = 0; i < formatters.length; i++) {
            var f = formatters[i];
            var axis = hcbc_getAxis(f.IsXAxis, f.AxisIndex, options);
            if (axis.labels == undefined) axis.labels = {};
            if (axis.custom == undefined) axis.custom = {};
            axis.custom.LabelFormatter = f;
            switch (f.Type) {
                case 'vol':
                    axis.labels.formatter = this.pointVol;
                    break;
                case 'future':
                    axis.labels.formatter = this.pointFuture;
                    break;
                case 'option':
                    axis.labels.formatter = this.pointOption;
                    break;
                case 'strike':
                    axis.labels.formatter = this.pointStrike;
                    break;
                case 'decimal':
                    axis.labels.formatter = this.pointDecimal;
                    break;
                case 'percent':
                    axis.labels.formatter = this.pointPercent;
                    break;
                case 'futureyield':
                    axis.labels.formatter = this.pointFutureYield;
                    break;
                case 'delta':
                    axis.labels.formatter = this.deltaLabel;
                    break;
                case 'greek':
                    switch (f.Greek) {
                        case 'D':
                            axis.labels.formatter = this.pointDelta;
                            break;
                        case 'G':
                            axis.labels.formatter = this.pointGamma;
                            break;
                        case 'V':
                            axis.labels.formatter = this.pointVega;
                            break;
                        case 'T':
                            axis.labels.formatter = this.pointTheta;
                            break;
                    }
                    break;
            }
            var product = this.Settings.Product;
            if (f.ProductIndex >= 0)
                product = this.Settings.Products[f.ProductIndex];
            axis.custom.Product = product;

        }
    },

    addIndicators: function(options) {
        var indicators = this.Settings.Indicators;
        if (indicators == null || indicators.length == 0)
            return;

        for (var i = 0; i < indicators.length; i++) {
            var ind = indicators[i];
            ind.marker = { enabled: false };

            switch (ind.type) {
                case 'sma':
                    ind.lineWidth = 1;
                    ind.enableMouseTracking = false;
                    ind.showInLegend = true;
                    ind.params = { period: ind.Period, index: ind.ValueIndex };
                    ind.name = ind.name !== undefined ? ind.name :  "MA " + ind.Period;
                    break;
                case 'vbp':
                    ind.enableMouseTracking = false;
                    ind.showInLegend = false;
                    ind.params = { volumeSeriesID: ind.VolumeSeriesID };
                    break;
            };

            //var linkedTo = options.series.find(function (series) { return series.id == indicator.linkedTo }); // IE11 doesn't support find()...ugh
            var linkedTo = options.series.filter(function (series) { return series.id == ind.linkedTo })[0];
            // not sure if this is valid for all cases, but for now this is what I am doing
            ind.tooltip = linkedTo.tooltip;
            //ind.custom = linkedTo.custom;
            ind.custom = $.extend({}, linkedTo.custom); // copy custom from linked series
            if (ind.LegendText !== undefined)
                ind.custom.LegendText = ind.LegendText;
            ind.yAxis = linkedTo.yAxis;
            options.series.push(ind);
        }

    },

    setTooltipPositioner: function (options) {

        var ttp = this.Settings.TooltipPositioner;
        if (ttp == null)
            return;

        var ttpFunction = null;
        switch (ttp.Type) {
            case 'static':
                ttpFunction = this.staticTooltip;
                break;
            case 'hover':
                // NOTE:  these values could also be set from properties added to the c# HoverTooltipPositioner class
                if (options.tooltip == undefined) options.tooltip = {};
                $.extend(options.tooltip,
                    {
                        padding: 8,
                        shadow: true,
                        borderWidth: 1,
                        shape: 'callout',
                        split: false,
                        backgroundColor: this.Settings.Theme.TooltipBG,
                        style: {
                            color: this.Settings.Theme.TooltipColor
                        }
                    });
                break;
        }

        if (ttpFunction != null) {
            if (options.tooltip == undefined) options.tooltip = {};
            options.tooltip.positioner = ttpFunction;
        }
    },

    setLegendPosition: function (options) {
        var lp = this.Settings.LegendPositioner;
        if (lp == null)
            return;

        if (options.legend == undefined) options.legend = {};
        if (options.legend.enabled != true)
            return;

        switch (lp.Type) {
            case 'topright':
                if (options.rangeSelector == undefined || options.rangeSelector.enabled == false)
                    lp.y = 0;
                break;
        }

        $.extend(options.legend, lp);

    },

    setLegendFormatter: function (options) {
        var lf = this.Settings.LegendFormatter;
        if (lf == null)
            return;

        var lfFunction = null;
        switch (lf.Type) {
            case 'standard':
                lfFunction = this.standardLegend;
                break;
        }

        if (lfFunction != null) {
            if (options.legend == undefined) options.legend = {};
            options.legend.labelFormatter = lfFunction;
        }
    },

    addPlotLines: function(options) {
        var plotLines = this.Settings.PlotLines;
        if (plotLines == null || plotLines.length == 0)
            return;

        for (var i = 0; i < plotLines.length; i++) {
            var pl = plotLines[i];

            var axis = hcbc_getAxis(pl.IsXAxis, pl.AxisIndex, options)
            if (axis.plotLines == undefined) axis.plotLines = [];
            axis.plotLines.push(pl);
        }
    },

    addPlotBands: function (options) {
        var plotBands = this.Settings.PlotBands;
        if (plotBands == null || plotBands.length == 0)
            return;

        for (var i = 0; i < plotBands.length; i++) {
            var pb = plotBands[i];
            var axis = hcbc_getAxis(pb.IsXAxis, pb.AxisIndex, options)
            if (axis.plotBands == undefined) axis.plotBands = [];
            axis.plotBands.push(pb);
        }
    },

    addExporting: function (options) {
        if (options.exporting == undefined) options.exporting = {};
        if (options.exporting.buttons == undefined) options.exporting.buttons = {};
        if (options.exporting.buttons.contextButton == undefined) options.exporting.buttons.contextButton = {};
        if (options.exporting.buttons.contextButton.menuItems == undefined)
            options.exporting.buttons.contextButton.menuItems = ['viewFullscreen', 'printChart', 'separator', 'downloadPNG', 'downloadJPEG', 'downloadPDF', 'downloadSVG'];

        if (this.Settings.AllowDataExport)
            options.exporting.buttons.contextButton.menuItems.push('downloadCSV');

        if (this.Settings.IncludeStats) {
            var statsDefinition = {};
            statsDefinition.text = 'Show Statistics';
            statsDefinition.onclick = this.showStats;

            if (options.exporting.menuItemDefinitions == undefined) options.exporting.menuItemDefinitions = {};
            options.exporting.menuItemDefinitions.stats = statsDefinition;

            options.exporting.buttons.contextButton.menuItems.push('stats');

            var statsCopyDefinition = {};
            statsCopyDefinition.text = 'Copy Statistics';
            statsCopyDefinition.onclick = this.copyStats;

            if (options.exporting.menuItemDefinitions == undefined) options.exporting.menuItemDefinitions = {};
            options.exporting.menuItemDefinitions.statsCopy = statsCopyDefinition;

            options.exporting.buttons.contextButton.menuItems.push('statsCopy');
        }
    },

    copyStats: function () {

        var cliptext = "Series\tMin\tMax\tMean\tLast\tSigma\tSigmaPlus\tSigmaMinus\tZScore\tLQ\tUQ\tMedian\n";

        this.series.forEach(function (s) {

            if (s.options.showInStats && s.visible && s.options.id != 'highcharts-navigator-series') {

                var data = s.points.filter(function (x) { return x.x >= s.xAxis.min && x.x <= s.xAxis.max; }).map(function (x) { return x.y; }); //x => x.x >= s.xAxis.min && x.x <= s.xAxis.max).map(x => x.y);
                var stats = new Stats(data);

                cliptext += s.name + "\t";
                cliptext += stats.min() + "\t";
                cliptext += stats.max() + "\t";
                cliptext += stats.mean() + "\t";
                cliptext += stats.last() + "\t";
                cliptext += stats.SD() + "\t";
                cliptext += stats.SDPlus() + "\t";
                cliptext += stats.SDMinus() + "\t";
                cliptext += stats.ZScore() + "\t";
                cliptext += stats.LQ() + "\t";
                cliptext += stats.UQ() + "\t";
                cliptext += stats.median();

                cliptext += "\n";
            }
        });

        var tb = $("<textarea>");
        $("body").append(tb);
        tb.val(cliptext).focus().select();
        document.execCommand("copy");
        tb.remove();

        alert('Statistics copied to the clipboard');

    },

    showStats: function () {

        $('#chartstats').remove();

        var settings = this.userOptions.custom.Settings;
        var chart = $("#" + settings.Id);
        chart.css({ position: 'relative' });


        var chartWidth = chart.width();
        var statsLeft = chartWidth - 325 - 20;

        var statsDiv = $("<div id='chartstats' class='qs-popup generic-popup card padding-md' style='width:325px;position:absolute;text-align:left;max-height:450px;overflow-y:auto;'><div>").appendTo(chart);
        statsDiv.css({ top: "16px", left: statsLeft + "px" });

        var statsDivHeader = $("<div class='header'>Chart Statistics</div>").appendTo(statsDiv);
        var statsDivBody = $("<div class='body'></div>").appendTo(statsDiv);

        this.series.forEach(function (s) {

            if (s.options.showInStats && s.visible && s.options.id != 'highcharts-navigator-series') {

                var data = s.points.filter(function (x) { return x.x >= s.xAxis.min && x.x <= s.xAxis.max; }).map(function (x) { return x.y; }); //x => x.x >= s.xAxis.min && x.x <= s.xAxis.max).map(x => x.y);
                var stats = new Stats(data);

                var card = $("<div class='card padding-md'></div>").appendTo(statsDivBody);
                var header = $("<div class='header'><div style='width:20px;height:3px;background-color:" + s.color + ";float:left;margin-right:5px;margin-top:5px;'></div>" + s.name + "</div>").appendTo(card);
                var body = $("<div class='body'></div>").appendTo(card);

                var table = $("<table class='w-lg' style='margin-bottom:0'></table>").appendTo(body);

                table.append("<tr><td>Min:</td><td class='number'>" + stats.format(stats.min(), s.yAxis) + "</td><td style='padding-left:2em'>&sigma;:</td><td class='number'>" + stats.format(stats.SD(), s.yAxis) + "</td><td style='padding-left:2em'>LQ:</td><td class='number'>" + stats.format(stats.LQ(), s.yAxis) + "</td></tr>");
                table.append("<tr><td>Max:</td><td class='number'>" + stats.format(stats.max(), s.yAxis) + "</td><td style='padding-left:2em'>+&sigma;:</td><td class='number'>" + stats.format(stats.SDPlus(), s.yAxis) + "</td><td style='padding-left:2em'>UQ:</td><td class='number'>" + stats.format(stats.UQ(), s.yAxis) + "</td></tr>");
                table.append("<tr><td>Mean:</td><td class='number'>" + stats.format(stats.mean(), s.yAxis) + "</td><td style='padding-left:2em'>-&sigma;:</td><td class='number'>" + stats.format(stats.SDMinus(), s.yAxis) + "</td><td style='padding-left:2em'>Median:</td><td class='number'>" + stats.format(stats.median(), s.yAxis) + "</td></tr>");
                table.append("<tr><td>Last:</td><td class='number'>" + stats.format(stats.last(), s.yAxis) + "</td><td style='padding-left:2em'>Z:</td><td class='number'>" + Highcharts.numberFormat(stats.ZScore(), 2) + "</td></tr>");
            }
        });
    },

    nullTooltip: function () {
        return '';
    },

    staticTooltip : function(width, height, point) {
        var chart = this.chart,
            position;

        if (point.isHeader) {
            position = {
                x: Math.max(
                    // Left side limit
                    chart.plotLeft,
                    Math.min(
                        point.plotX + chart.plotLeft - width / 2,
                        // Right side limit
                        chart.chartWidth - width - chart.marginRight
                    )
                ),
                y: point.plotY
            };
        } else {

            var ttp = chart.userOptions.custom.Settings.TooltipPositioner;
            var yAxisTop = (point.series == undefined || (ttp != null && ttp.ForceY0)) ? chart.yAxis[0].top : point.series.yAxis.top;

            var yPlotAdjust = chart.plotTop;
            if (chart.hoverSeries != undefined && chart.hoverSeries.userOptions.TooltipUseChartCoordinatesY) // hack for bubble chart, others?
                yPlotAdjust = 0;

            position = {
                x: chart.plotLeft,
                y: yAxisTop - yPlotAdjust + (ttp == null ? 0 : ttp.YAdjust)
            };
        }
        return position;
    },

    standardLegend: function () {
        return (this.options.custom == null || this.options.custom.LegendText == null) ? this.name : this.options.custom.LegendText;
    },

    pointVol: function () {
        return hcbc_volatilityConvert(this.value, this.axis.userOptions.custom.Product.VolatilityFormat);
    },

    pointFuture: function () {
        return hcbc_futureConvert(this.value, this.axis.userOptions.custom.Product.FutureFormat);
    },

    pointOption: function () {
        return hcbc_optionConvert(this.value, this.axis.userOptions.custom.Product.OptionFormat);
    },

    pointStrike: function () {
        return hcbc_strikeConvert(this.value, this.axis.userOptions.custom.Product.StrikeFormat);
    },

    pointDelta: function () {
        return hcbc_greekConvert(this.value, this.axis.userOptions.custom.Product.DeltaFormat);
    },

    pointGamma: function () {
        return hcbc_greekConvert(this.value, this.axis.userOptions.custom.Product.GammaFormat);
    },

    pointVega: function () {
        return hcbc_greekConvert(this.value, this.axis.userOptions.custom.Product.VegaFormat);
    },

    pointTheta: function () {
        return hcbc_greekConvert(this.value, this.axis.userOptions.custom.Product.ThetaFormat);
    },

    deltaLabel: function () {

        if (this.value < 50)
            return Highcharts.numberFormat(this.value, 0) + 'P';
        else if (this.value > 50)
            return Highcharts.numberFormat(100 - this.value, 0) + 'C';
        else
            return Highcharts.numberFormat(this.value, 0);
    },

    pointDecimal: function () {
        return Highcharts.numberFormat(this.value, this.axis.userOptions.custom.LabelFormatter.Decimals);
    },

    pointPercent: function () {
        return Highcharts.numberFormat(this.value * 100.0, this.axis.userOptions.custom.LabelFormatter.Decimals) + '%';
    },

    pointFutureYield: function () {

        var m = this.axis.userOptions.custom.LabelFormatter.Slope;
        var b = this.axis.userOptions.custom.LabelFormatter.Intercept;
        var s1 = hcbc_futureConvert(this.value, this.axis.userOptions.custom.Product.FutureFormat);
        var y = m * this.value + b;
        var s2 = Highcharts.numberFormat(100 * y, this.axis.userOptions.custom.LabelFormatter.Decimals) + '%';

        return s1 + "<br/>" + s2;
    },

    destroyChart: function (chart) {
        var charts = Highcharts.charts;
        for (var i = 0; i < charts.length; i++) {
            if (charts[i] == chart) {
                chart.destroy();
                charts.splice(i, 1);
                return;
            }
        }

    },

    dispose: function () {

        this.destroyChart(this.highchart);

        $(window).unbind('ScriptLoaded', this.fnCreateChart);

        QuikStrike.Highcharts.ChartControl.callBaseMethod(this, 'dispose');

    }

}

function hcbc_addWatermark(chart) {

    if (chart.custom == null)
        chart.custom = {};
    else {
        if (chart.custom.imgWatermark != undefined)
            chart.custom.imgWatermark.destroy();
        if (chart.custom.imgWatermark2 != undefined)
            chart.custom.imgWatermark2.destroy();
    }

    var settings = chart.userOptions.custom.Settings;
    // NOTE: the image width/height is required for this to work during exporting
    chart.custom.imgWatermark = chart.renderer.image(settings.WatermarkImg).add();

    switch (settings.WatermarkPos) {
        case 'TL':
            chart.custom.imgWatermark.attr({
                x: chart.plotLeft + 7 + settings.WatermarkX,
                y: chart.plotTop + 2 + settings.WatermarkY,
                width: 56,
                height: 80
            });
            break;
        default:
            chart.custom.imgWatermark.attr({
                x: chart.plotWidth + chart.plotLeft - (56 + 7) + settings.WatermarkX,
                y: chart.plotTop + 2 + settings.WatermarkY,
                width: 56,
                height: 80
            });
            break;
    }

                });
                break;
            case 'BR':
                chart.custom.imgWatermark2.attr({
                    x: chart.plotWidth + chart.plotLeft - (settings.Watermark2Size.Width + 10) + settings.Watermark2X,
                    y: chart.plotHeight + chart.plotTop - settings.Watermark2Size.Height - 10 + settings.Watermark2Y,
                    width: settings.Watermark2Size.Width,
                    height: settings.Watermark2Size.Height
                });
                break;
            default:
                chart.custom.imgWatermark2.attr({
                    x: chart.plotWidth + chart.plotLeft - (56 + 7 + settings.Watermark2Size.Width + 5) + settings.Watermark2X,
                    y: chart.plotTop + 2 + settings.Watermark2Y,
                    width: settings.Watermark2Size.Width,
                    height: settings.Watermark2Size.Height
                });
                break;
        }
    }

}

function hcbc_decimalFormat(value, precision) {
    return value.toFixed(precision);

function hcbc_futureConvert(value, priceFormat) {
    return hcbc_toFixed(hcbc_baseConvert(value, priceFormat), priceFormat);
}

    return hcbc_toFixed((hcbc_baseConvert(value, optionFormat) * optionFormat.ScaleFactor), optionFormat);

function hcbc_toFixedOld(value, format) {

    var ret = value.localeFormat('N' + precision);
    if (!useSeparator) {
        var sep = Highcharts.getOptions().lang.thousandsSep;
        return ret.replace(new RegExp('\\' + sep, 'g'), '');
    }
    return ret;
}

function hcbc_baseConvert(value, numFormat) {

    var decimalBase = numFormat.DecimalBase;

    if (decimalBase != 100) {
        var sign = value < 0 ? -1 : 1;
        value = Math.abs(value);
        var left = Math.floor(value);
        var right = value - left;
        var scaleFactor = decimalBase <= 10 ? 10 : 100;
        value = sign * (left + right * decimalBase / scaleFactor);
    }
    return value;
}

function hcbc_getPlotLine(axis, id) {
    for (var i = 0; i < axis.options.plotLines.length; i++)
        if (axis.options.plotLines[i].id == id)
            return axis.options.plotLines[i];
    return null;
}

function hcbc_getAxis(isX, index, options) {
    if (isX) {
        if (options.xAxis == undefined) options.xAxis = [];
        if (!Array.isArray(options.xAxis))
            // if the xAxis is not an array then return it regardless of index
            return options.xAxis;
        if (options.xAxis[index] == undefined) options.xAxis[index] = {};
        return options.xAxis[index];
    }
    else {
        if (options.yAxis == undefined) options.yAxis = [];
        if (!Array.isArray(options.yAxis))
            // if the yAxis is not an array then return it regardless of index
            return options.yAxis;
        if (options.yAxis[index] == undefined) options.yAxis[index] = {};
        return options.yAxis[index];
    }
}

            dstCharts[i].xAxis[xIndex].setExtremes(srcChart.xAxis[xIndex].min, srcChart.xAxis[xIndex].max, true, true, { purpose: 'sync' });
        }
}

/* Stats Class */
function Stats(data) {
    this.data = data;
}

Stats.prototype.min = function () {

    if (this.data.length === 0) return 0;
    return Math.min.apply(null, this.data);
}

Stats.prototype.max = function() {

    if (this.data.length === 0) return 0;
    return Math.max.apply(null, this.data);
}

Stats.prototype.mean = function() {

    if (this.data.length === 0) return 0;
    return this.data.reduce(function (a, b) { return a + b; }) / this.data.length;
}

Stats.prototype.median = function() {

    if (this.data.length === 0) return 0;

    //make a copy so we don't change the original

    var values = this.data.slice(); //[...this.data];

    values.sort(function (a, b) { return a - b; });

    var half = Math.floor(values.length / 2);

    if (values.length % 2)
        return values[half];

    return (values[half - 1] + values[half]) / 2.0;
}

Stats.prototype.last = function() {

    if (this.data.length === 0) return 0;

    return this.data[this.data.length - 1];
}

Stats.prototype.SD = function() {

    if (this.data.length === 0) return 0;

    const n = this.data.length;
    const mean = this.data.reduce(function (a, b) { return a + b; }) / n
    return Math.sqrt(this.data.map(function (x) { return Math.pow(x - mean, 2); }).reduce(function (a, b) { return a + b; }) / n)
}

Stats.prototype.SDPlus = function() {
    return this.mean() + this.SD();
}

Stats.prototype.SDMinus = function() {
    return this.mean() - this.SD();
}

Stats.prototype.ZScore = function() {
    return (this.last() - this.mean()) / this.SD();
}

Stats.prototype.UQ = function() {
    return this.quartile(this.data, 0.75);
}

Stats.prototype.LQ = function() {
    return this.quartile(this.data, 0.25);
}

Stats.prototype.quartile = function(data, q) {

    if (data.length === 0) return 0;

    var values = data.slice(); //[...data];
    values.sort(function (a, b) { return a - b; });

    var pos = ((values.length) - 1) * q;
    var base = Math.floor(pos);
    var rest = pos - base;
    if ((values[base + 1] !== undefined)) {
        return values[base] + rest * (values[base + 1] - values[base]);
    } else {
        return values[base];
    }
}


Stats.prototype.format = function(value, axis) {

    var decimals = Math.abs(value) < 1000 ? 2 : 0;

    if (axis.options == undefined)
        return Highcharts.numberFormat(value, decimals);

    if (axis.options.custom == undefined)
        return Highcharts.numberFormat(value, decimals);

    if (axis.options.custom.LabelFormatter == undefined)
        return Highcharts.numberFormat(value, decimals);

    if (axis.options.custom.LabelFormatter.Type == undefined)
        return Highcharts.numberFormat(value, decimals);

    switch (axis.options.custom.LabelFormatter.Type) {
        case 'vol':
            return hcbc_volatilityConvert(value, axis.options.custom.Product.VolatilityFormat);

        case 'future':
            return hcbc_futureConvert(value, axis.userOptions.custom.Product.FutureFormat);

        case 'option':
            return hcbc_optionConvert(value, axis.userOptions.custom.Product.OptionFormat);

        case 'strike':
            return hcbc_strikeConvert(value, axis.userOptions.custom.Product.StrikeFormat);

        case 'decimal':
            return Highcharts.numberFormat(value, axis.userOptions.custom.LabelFormatter.Decimals);

        case 'percent':
            return Highcharts.numberFormat(value * 100.0, axis.userOptions.custom.LabelFormatter.Decimals) + '%';
    }

    return Highcharts.numberFormat(value, decimals);
}
