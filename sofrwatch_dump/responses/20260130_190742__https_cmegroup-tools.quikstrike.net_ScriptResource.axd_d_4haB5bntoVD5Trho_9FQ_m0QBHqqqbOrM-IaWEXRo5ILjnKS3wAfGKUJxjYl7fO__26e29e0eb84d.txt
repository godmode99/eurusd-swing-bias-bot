if (Sys.WebForms != null)
    Sys.WebForms.PageRequestManager.getInstance().add_endRequest(popupNav_EndRequestHandler);

function popupNav_EndRequestHandler(sender, args)
{
    if (args.get_error() == undefined)
    {
        pp_documentReady();
    }
}

$(document).ready(function ()
{
    pp_documentReady();
});

var pp_data = null;
var pp_needsfill = true;
var pp_showselected = true;

function pp_documentReady()
{
    pp_data = null;
    pp_needsfill = true;
    pp_showselected = true;

    $('#lnkFilterTrigger').unbind('click');
    $('#lnkFilterTrigger').click(function (event)
    {
        if ($('#divFilterPopup').is(':visible'))
            pp_close();
        else
            pp_open();
        event.preventDefault();
    });

    $('#lnkFilterClear').unbind('click');
    $('#lnkFilterClear').click(function (event)
    {
        var data = pp_getData();

        $('#txtFilterText').val('');
        if (data.SingleMode || $('#divFilterPopup').is(':visible'))
        {
            pp_needsfill = true;
            pp_showselected = true;
            pp_fill();
        }
        else
        {
            // clear selected in DTO
            data.Selected = [];
            pp_setData(data);

            // uncheck all in UI
            $('#tblFilterProducts input[type="checkbox"]').prop('checked', false);

            // submit
            $('#btnFilterApply').click();
        }
        // force watermark extender to add watermark
        $('#txtFilterText').focus();
        $('#txtFilterText').blur();
        event.preventDefault();
    });

    $('#btnFilterApply').unbind('click');
    $('#btnFilterApply').click(function (event)
    {
        pp_setSelected();
    });

    $('#divFilterPopup a.check-all').unbind('click');
    $('#divFilterPopup a.check-all').click(function (event)
    {
        $('#tblFilterProducts input[type="checkbox"]').prop('checked', true);
        event.preventDefault();
    });

    $('#divFilterPopup a.uncheck-all').unbind('click');
    $('#divFilterPopup a.uncheck-all').click(function (event)
    {
        $('#tblFilterProducts input[type="checkbox"]').prop('checked', false);
        event.preventDefault();
    });

    $(document).off('click', '#divFilterPopup a.p-trig');
    $(document).on('click', '#divFilterPopup a.p-trig', function (event)
    {
        // fire postback
        __doPostBack($('#divFilterPopup').attr('parentid'), $(this).closest('tr').attr('id'));
        event.preventDefault();
        //event.stopPropagation();
    });

    $(document).off('click', '#tblFilterProducts a.checkon');
    $(document).on('click', '#tblFilterProducts a.checkon', function (event)
    {
        var grp = $(this).closest('tr[grp]').attr('grp');
        $('#tblFilterProducts input[type="checkbox"][grp="' + grp + '"]').prop('checked', true);
        event.preventDefault();
    });

    $(document).off('click', '#tblFilterProducts a.checkoff');
    $(document).on('click', '#tblFilterProducts a.checkoff', function (event)
    {
        var grp = $(this).closest('tr[grp]').attr('grp');
        $('#tblFilterProducts input[type="checkbox"][grp="' + grp + '"]').prop('checked', false);
        event.preventDefault();
    });

    $('#txtFilterText').unbind('keyup');
    $('#txtFilterText').keyup(function (event)
    {
        // ignore enter key
        if (event.which == 13)
            event.preventDefault();
        else
            pp_processKey();
    });

    $('#txtFilterText').unbind('keydown');
    $('#txtFilterText').keydown(function (event)
    {
        // handle enter key by selecting everything so far and closing
        if (event.which == 13)
        {
            // if the dropdown is not open, then simply open it
            if (!$('#divFilterPopup').is(':visible'))
                pp_open();
            else if (!pp_isTextEmpty())
            {
                if ($("#tblFilterProducts input:checkbox").length > 0) {
                    // check all in UI
                    $('#tblFilterProducts input[type="checkbox"]').prop('checked', true);

                    // update DTO
                    pp_setSelected();

                    // submit
                    $('#btnFilterApply').click();
                }
                else if ($("#tblFilterProducts a.p-trig").length == 1) {
                    $("#tblFilterProducts a.p-trig").click();
                }
            }

            event.preventDefault();

        }
    });

    $(document).click(function (event)
    {
        if ($(event.target).closest('#divFilterPopup, div.productpicker-text').length == 0)
            pp_close();
    });

    //$('#txtFilterText').focus();

}

function pp_processKey()
{
    pp_needsfill = true;
    pp_showselected = pp_isTextEmpty();
    pp_open();
}

function pp_getData()
{
    if (pp_data == null)
    {
        var jsonString = $('#hdnFilterData').val();
        pp_data = $.parseJSON(jsonString);
    }
    return pp_data;
}

function pp_setData(data)
{
    pp_data = data;
    $('#hdnFilterData').val(JSON.stringify(pp_data));
}

function pp_isTextEmpty()
{
    var str = $('#txtFilterText').val();
    //if (str.length == 0 || str.toLowerCase().includes('enter search text'))
    if (str.length == 0 || str.toLowerCase().indexOf('enter search text') >= 0)
        return true;
    else
        return false;
}

function pp_fill()
{
    if (!pp_needsfill)
        return;

    var data = pp_getData();

    $('#divFilterEmpty').hide();
    $('#tblFilterProducts').show();
    if (data.SingleMode)
        $('#divFilterButtons').hide();
    else
        $('#divFilterButtons').show();

    pp_needsfill = false;

    var str = '';
    if (!pp_isTextEmpty())
        str = $('#txtFilterText').val().trim().toLowerCase();

    $("#tblFilterProducts > tbody").empty();
    var table = $('#tblFilterProducts');
    var prod = 0;
    var count = 0;
    var product = data.Products.length > 0 ? data.Products[prod++] : null;
    for (grp = 0; grp < data.Groups.length; grp++)
    {
        var needsHeader = true;
        do
        {
            if (pp_filter(str, product))
            {
                if (needsHeader)
                {
                    needsHeader = false;
                    if (data.SingleMode)
                        table.append("<tr class='heading2' grp='" + grp + "'><th>Clr</th><th>Gbx</th><th>Prs</th><th>QS</th><th><span style='position:relative;top:1px'>" + data.Groups[grp] + "</span></th></tr>");
                    else
                        table.append("<tr class='heading' grp='" + grp + "'><th colspan='6'><a href='#' class='checkon'><img align='absmiddle' src='../../images/icons/checkbox_yes.png'></a> <a href='#' class='checkoff'><img align='absmiddle' src='../../images/icons/checkbox_no.png'></a>   <span style='position:relative;top:1px'>" + data.Groups[grp] + "</span></th></tr>");
                    //table.append("<tr class='col-heading'><th/><th>Clr</th><th>Glx</th><th>Product</th></tr>");
                }
                count++;
                if (data.SingleMode)
                    table.append("<tr class='" + (product.Id == data.SelectedId ? "selected" : "") + "' id='" + product.Id + "'><td>" + product.Clear + "</td><td>" + product.Globex + "</td><td>" + product.Prs + "</td><td>" + product.QS + "</td><td><a class='p-trig' href='#'>" + product.Name + "</a></td></tr>");
                else
                    table.append("<tr id='" + product.Id + "'><td class='check'><input grp='" + grp + "' type='checkbox'" + pp_addchecked(data, product) + "></td><td>" + product.Clear + "</td><td>" + product.Globex + "</td><td>" + product.Prs + "</td><td>" + product.Fut + "</td><td>" + product.Name + "</td></tr>");
            }
            product = prod < data.Products.length ? data.Products[prod++] : null;

        } while (product != null && product.Group == grp)
    }

    if (count == 0)
    {
        $('#divFilterEmpty').show();
        $('#divFilterEmpty').text(str.length < 3 ? 'Please enter additional search text to continue' : 'There are no products that meet the filter criteria');
        $('#tblFilterProducts').hide();
        $('#divFilterButtons').hide();
    }
}

function pp_filter(str, product)
{
    if (str.length == 0)
        return true;

    var orClauses = str.split(',');

    for (var i = 0; i < orClauses.length; i++)
    {
        if (pp_checkClause(orClauses[i], product))
            return true;
    }

    return false;
}

function pp_checkClause(clause, product)
{
    if (clause.length == 0)
        return false;

    var tokens = clause.split(' ');
    for (var j = 0; j < tokens.length; j++)
    {
        if (!pp_check(tokens[j], product, j > 0))
            return false;
    }

    return true;
}

function pp_check(str, product, ignorelength)
{
    var len = str.length;

    if (len == 0)
        return true;

    if (len <= 3 && (str == product.Globex.toLowerCase() || str == product.Clear.toLowerCase() || str == product.Prs.toLowerCase() || str == product.Fut.toLowerCase() || str == product.QS.toLowerCase()))
        return true;

    //if ((len > 2 || ignorelength) && product.Name.toLowerCase().includes(str))
    if ((len > 2 || ignorelength) && product.Name.toLowerCase().indexOf(str) >= 0)
        return true;

    return false;
}

function pp_setSelected()
{
    var data = pp_getData();

    data.Selected = [];

    $("#tblFilterProducts input:checkbox:checked").each(function()
    {
        data.Selected.push(parseInt($(this).closest('tr[id]').attr('id')));
    });

    if (data.Selected.length == data.Products.length)
        data.Selected = [];

    pp_setData(data);

}

function pp_addchecked(data, product)
{
    //if (!pp_showselected || data.Selected == null)
    //    return '';
    //return data.Selected.indexOf(product.Id) < 0 ? '' : ' checked="checked" ';
    if (!pp_showselected)
        return '';
    return data.Selected == null || data.Selected.length == 0 || data.Selected.indexOf(product.Id) >= 0 ? ' checked="checked" ' : '';
}

function pp_open()
{
    var popup = $('#divFilterPopup');
    var anchor = $('#txtFilterText');
    var anchorOffset = anchor.offset();
    var pos = [anchorOffset.left, anchorOffset.top + anchor.outerHeight()];
    if (popup.parent().attr('id') != '#upMain')
        popup.appendTo($('#upMain'));
    pp_fill();
    popup.show();
    popup.css({ left: pos[0], top: pos[1] });
}

function pp_close()
{
    $('#divFilterPopup').hide();
}
