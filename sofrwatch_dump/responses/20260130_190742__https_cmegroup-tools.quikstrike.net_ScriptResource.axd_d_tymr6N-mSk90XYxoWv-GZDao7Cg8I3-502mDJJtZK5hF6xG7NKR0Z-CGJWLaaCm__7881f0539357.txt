if (window.Sys !== undefined && Sys.WebForms != null)
    Sys.WebForms.PageRequestManager.getInstance().add_endRequest(hovitm_EndRequestHandler);

function hovitm_EndRequestHandler(sender, args)
{
    if (args.get_error() == undefined)
    {
        hovitm_Init();
    }
}

$(document).ready(function ()
{
    hovitm_Init();
});

function hovitm_Init()
{

    //$("div.hover-container div.hover-item").each(function ()
    //{
    //    global_centerV($(this));
    //});

    $(document).off('mouseover', 'div.hover-container .hover-parent');
    $(document).on('mouseover', 'div.hover-container .hover-parent', function (event)
    {
        var hoverItem = $(this).children('div.hover-item');
        hoverItem.show();
        global_centerV(hoverItem);
        hoverItem.css({ left: $(this).width() - hoverItem.outerWidth() });
    });

    $(document).off('mouseout', 'div.hover-container .hover-parent');
    $(document).on('mouseout', 'div.hover-container .hover-parent', function (event)
    {
        $(this).children('div.hover-item').hide();
    });

}
